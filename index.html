function loadProfessionalData() { const profId = getQueryParam('prof'); const professionals = JSON.parse(localStorage.getItem('nail_professionals') || '[]'); if (profId) { activeProf = professionals.find(p => p.id === profId); } else if (professionals.length > 0) { activeProf = professionals[0]; } if (!activeProf) { document.getElementById('prof-header').innerText = "Profissional não encontrado"; document.getElementById('prof-address-display').innerText = "--"; return; } document.getElementById('prof-header').innerText = activeProf.name; document.getElementById('prof-address-display').innerText = activeProf.address || "Atendimento a Domicílio"; const serviceSelect = document.getElementById('service-select'); serviceSelect.innerHTML = ''; if (activeProf.services) { activeProf.services.forEach(s => { const opt = document.createElement('option'); opt.value = JSON.stringify(s); opt.innerText = `${s.name} - R$ ${parseFloat(s.price).toFixed(2).replace('.', ',')}`; serviceSelect.appendChild(opt); }); } // Define a data do campo como HOJE por padrão const dateInput = document.getElementById('booking-date'); const today = new Date().toISOString().split('T')[0]; dateInput.value = today; dateInput.setAttribute('min', today); // Renderiza os horários imediatamente ao carregar a página renderTimeSlots(); } function renderTimeSlots() { const slotsGrid = document.getElementById('slots-container'); slotsGrid.innerHTML = ''; selectedSlot = null; const selectedDate = document.getElementById('booking-date').value; if (!selectedDate) { slotsGrid.innerHTML = '

Selecione uma data para ver os horários.

'; return; } // Lista padrão de horários do profissional const defaultSlots = ["08:00", "09:00", "10:00", "11:00", "13:00", "14:00", "15:00", "16:00", "17:00"]; const baseSlots = activeProf && activeProf.availableSlots && activeProf.availableSlots.length > 0 ? activeProf.availableSlots : defaultSlots; // Busca agendamentos existentes no localStorage const appointments = JSON.parse(localStorage.getItem('nail_appointments') || '[]'); // Filtra os horários já ocupados para a data e profissional selecionados const bookedSlots = appointments .filter(app => app.profId === activeProf.id && app.datetime.startsWith(selectedDate)) .map(app => app.datetime.split('T')[1]); // Exibe apenas os horários livres const availableSlots = baseSlots.filter(time => !bookedSlots.includes(time)); if (availableSlots.length === 0) { slotsGrid.innerHTML = '

Nenhum horário disponível para esta data.

'; return; } availableSlots.forEach(time => { const btn = document.createElement('div'); btn.className = 'slot-btn'; btn.innerText = time; btn.onclick = function() { document.querySelectorAll('.slot-btn').forEach(b => b.classList.remove('selected')); btn.classList.add('selected'); selectedSlot = time; }; slotsGrid.appendChild(btn); }); }
