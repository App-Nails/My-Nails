async function confirmSlot(id) { let slots = getDB('slots'); let idx = slots.findIndex(s => s.id === id); if (idx !== -1) { const slot = slots[idx]; slot.booked = true; setDB('slots', slots); // Busca o agendamento no Supabase pelo horário e pro_id para pegar os dados do serviço e preço corretos const { data: existingBookings } = await _supabase .from('agendamentos') .select('*') .eq('pro_id', slot.proPhone) .like('horario', `%${slot.time}%`); let serviceName = 'Serviço Padrão'; let servicePrice = 50.00; // Valor padrão de fallback if (existingBookings && existingBookings.length > 0) { const booking = existingBookings[0]; if (booking.servico) serviceName = booking.servico; if (booking.preco) servicePrice = parseFloat(booking.preco); } const { error } = await _supabase.from('agendamentos').insert([ { pro_id: slot.proPhone, nome: existingBookings?.[0]?.nome || 'Cliente Confirmado', telefone: existingBookings?.[0]?.telefone || '', horario: `${slot.date} ${slot.time}`, servico: serviceName, preco: servicePrice } ]); if (error) { console.error("Erro ao salvar no Supabase:", error); } renderSlots(); await renderBookingsPro(); await calculateFinances(); } }
