// Adicione este trecho no script do profissional.html setInterval(() => { const appointments = JSON.parse(localStorage.getItem('nail_appointments') || '[]'); const now = new Date().getTime(); appointments.forEach(app => { const appTime = new Date(app.datetime).getTime(); const diffMinutes = (appTime - now) / (1000 * 60); if (diffMinutes > 0 && diffMinutes <= 30 && !app.alert30Sent) { if ("Notification" in window && Notification.permission === "granted") { new Notification("⏰ Próximo Atendimento em 30 min!", { body: `Cliente: ${app.client} - Serviço: ${app.service}` }); } else { alert(`⏰ Atenção: Faltam 30 minutos para o atendimento de ${app.client}!`); } app.alert30Sent = true; } }); localStorage.setItem('nail_appointments', JSON.stringify(appointments)); }, 10000);
